Add .env file with the following keys

## SES CREDENTIALS ##
SNS_HOST = YOUR SES HOST
SNS_USER = YOUR SES USER
SNS_PASS = YOUR SES PASSWORD
EMAIL_FROM = YOUR SES EMAIL ID

## DATABASE CREDENTIALS ##
DB_HOST = YOUR DB HOST NAME
DB_USER = YOUR DB USER
DB_PASS = YOUR DB PASSWORD
DB_NAME = YOUR DB NAME

## PROKERALA API CREDENTIALS ##
CLIENT_ID = YOUR PROKERALA CLIENT ID
CLIENT_SECRET = YOUR PROKERALA CLIENT SECRET
GRANT_TYPE = YOUR PROKERALA GRANT TYPE
GENERATE_TOKEN_API = YOUR PROKERALA TOKEN API END POINT

## AUTH CREDENTIAL ##
API_KET = YOUR APPLICATION AUTH API KEY
API_SECRET = YOUR APPLICATION AUTH SECRET

## MESSAGE API CREDENTIALS ##
MESSAGE_USERNAME = YOUR SMS API USER NAME
MESSAGE_PASSWORD = YOUR SMS API PASSWORD
MESSAGE_SENDER_ID = YOUR SMS API SENDER ID
MESSAGE_ROUTE = YOUR SMS API SENDER ROUTE

## CASHFREE API CREDENTIALS ##
KYC_END_POINT = YOUR CASHFREE END POINT
KYC_CLIENT_ID = YOUR CASHFREE CLIENT ID
KYC_CLIENT_SECRET = YOUR CASHFREE SECRET

## CCAVENUE API CREDENTIALS ##
WORKING_KEY = YOUR CCAVENUE WORKING KEY
MERCHANT_ID = YOUR CCAVENUE MERCHANT ID
ACCESS_CODE = YOUR CCAVENUE ACCESS CODE
CCAVENUE_URL = YOUR CCAVENUE END POINT
## END ##
